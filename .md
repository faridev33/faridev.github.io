# Langkah 1: Cek file apa saja yang telah diubah (opsional)
git status

# Langkah 2: Tandai semua file yang baru diubah/ditambahkan

git commit -m "Update portofolio: Tambah proyek AI CV Screening & update keahlian"

# Langkah 4: Upload/Push perubahan ke GitHub
git push origin main
